    function updateDropdowns(preserveSelection = true) {
        const selCur = document.getElementById('currFlavor');
        const selNxt = document.getElementById('nextFlavor');
        
        const curVal = selCur.value;
        const nxtVal = selNxt.value;
        
        selCur.innerHTML = '';
        selNxt.innerHTML = '';
        
        const selectedCatIndex = document.getElementById('productLine').value;
        if(database[selectedCatIndex]) {
            database[selectedCatIndex].items.forEach(item => {
                const displayName = item.name[currentLang];
                selCur.appendChild(new Option(displayName, item.id));
                selNxt.appendChild(new Option(displayName, item.id));
            });
        }

        if (preserveSelection && curVal && nxtVal && Array.from(selCur.options).some(o => o.value === curVal)) {
            selCur.value = curVal;
            selNxt.value = nxtVal;
        } else {
            document.getElementById('resultBox').style.display = 'none';
        }
    }

    function checkWash() {
        const selCur = document.getElementById('currFlavor');
        const selNxt = document.getElementById('nextFlavor');
        
        const currItem = flatList.find(f => f.id === parseInt(selCur.value));
        const nextItem = flatList.find(f => f.id === parseInt(selNxt.value));
        const box = document.getElementById('resultBox');
        const title = document.getElementById('resTitle');
        const desc = document.getElementById('resDesc');
        const t = i18n[currentLang];

        box.style.animation = 'none'; box.offsetHeight; 
        box.style.animation = 'slideDown 0.3s ease-out forwards';
        box.style.display = 'block';

        let innerTitle = ""; let innerDesc = ""; let iconHtml = "";
        const currAllergens = currItem.allergens || [];
        const nextAllergens = nextItem.allergens || [];
        const dangerAllergens = currAllergens.filter(a => !nextAllergens.includes(a));

        if (currItem.id === nextItem.id) {
            box.className = 'no-wash'; 
            iconHtml = `<svg width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round"><path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"></path><polyline points="22 4 12 14.01 9 11.01"></polyline></svg>`;
            innerTitle = iconHtml + t.sameT; innerDesc = t.sameD;
        } else if (currItem.isKabab || currItem.isYogurt || dangerAllergens.length > 0) {
            box.className = 'red-wash';
            iconHtml = `<svg width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round"><circle cx="12" cy="12" r="10"></circle><line x1="12" y1="8" x2="12" y2="12"></line><line x1="12" y1="16" x2="12.01" y2="16"></line></svg>`;
            if (currItem.isKabab) { innerTitle = iconHtml + t.kababT; innerDesc = t.kababD; }
            else if (currItem.isYogurt) { innerTitle = iconHtml + t.yogurtT; innerDesc = t.yogurtD; }
            else { innerTitle = iconHtml + t.blockT; innerDesc = t.blockD; }
        } else {
            box.className = 'air-wash';
            iconHtml = `<svg width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2.5" stroke-linecap="round"><path d="M9.59 4.59A2 2 0 1 1 11 8H2m10.59 11.41A2 2 0 1 0 14 16H2m15.73-8.27A2.5 2.5 0 1 1 19.5 12H2"></path></svg>`;
            if (currItem.isFF) { innerTitle = iconHtml + t.ffT; innerDesc = t.ffD; } 
            else { innerTitle = iconHtml + t.airT; innerDesc = t.airD; }
        }

        if (dangerAllergens.length > 0 && currItem.id !== nextItem.id) {
            const mappedNames = dangerAllergens.map(a => t['al_name_' + a]).join('، ');
            innerDesc += `<div class="allergen-warning"><strong><svg width='18' height='18' viewBox='0 0 24 24' fill='none' stroke='currentColor' stroke-width='2.5' stroke-linecap='round' stroke-linejoin='round'><circle cx='12' cy='12' r='10'></circle><line x1='12' y1='8' x2='12' y2='12'></line><line x1='12' y1='16' x2='12.01' y2='16'></line></svg> ${t.warn_title}</strong> ${mappedNames}</div>`;
        }

        const formatAl = (item) => {
            if (item.isFF) return `<span class="al-tag safe">${t.txtNoAllergens}</span>`;
            if (item.allergens && item.allergens.length > 0) {
                return `<div class="al-tags">` + item.allergens.map(a => `<span class="al-tag">${t['al_name_' + a]}</span>`).join('') + `</div>`;
            }
            return `<span class="al-tag safe">${t.txtNoAllergens}</span>`;
        };

        let compareHtml = `
            <div class="allergen-compare-box">
                <div class="al-row">
                    <strong>${t.lblCurrAl}</strong> 
                    <div class="flavor-name-display">${currItem.name[currentLang]}</div>
                    ${formatAl(currItem)}
                </div>
                <div class="al-row" style="margin-top: 15px;">
                    <strong>${t.lblNextAl}</strong> 
                    <div class="flavor-name-display">${nextItem.name[currentLang]}</div>
                    ${formatAl(nextItem)}
                </div>
        `;

        title.innerHTML = innerTitle;
        desc.innerHTML = innerDesc + compareHtml + `</div>`;
    }

    function saveNewFlavor(event) {
        event.preventDefault();
        
        const gIndex = document.getElementById('newGroupSelect').value;
        const nameKu = document.getElementById('newNameKu').value.trim();
        const nameAr = document.getElementById('newNameAr').value.trim();
        const nameEn = document.getElementById('newNameEn').value.trim();
        
        const isFF = document.getElementById('newFF').checked;
        const isKabab = document.getElementById('newKabab').checked;
        const isYogurt = document.getElementById('newYogurt').checked;
        
        const allergens = [];
        document.querySelectorAll('.al-check:checked').forEach(cb => {
            allergens.push(cb.value);
        });

        let maxId = -1;
        flatList.forEach(item => { if(item.id > maxId) maxId = item.id; });
        const newId = maxId + 1;

        const newItem = {
            id: newId,
            name: { ku: nameKu, ar: nameAr, en: nameEn },
            isFF: isFF,
            isKabab: isKabab,
            isYogurt: isYogurt,
            allergens: allergens
        };

        // Add to database
        database[gIndex].items.push(newItem);
        
        // Save to LocalStorage
        localStorage.setItem('qc_database_v2', JSON.stringify(database));
        
        // Rebuild and close
        buildLists();
        updateDropdowns(true);
        closeModal();
        
        // Success message
        alert(currentLang === 'ku' ? "تامەکە بە سەرکەوتوویی زیادکرا و لە مۆبایلەکەت سەیڤ بوو!" : "تمت إضافة النكهة وحفظها بنجاح!");
    }
